create table public.rental_reservations (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  total_price real null,
  deposit real null,
  caution real null,
  reservation_status public.ReservationStatus null,
  delivery_fees real null,
  delivery_address text null,
  customer_order_id integer null,
  constraint reservations_pkey primary key (id),
  constraint reservations_id_key unique (id),
  constraint rental_reservations_customer_order_id_fkey foreign KEY (customer_order_id) references customer_orders (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;

create trigger update_reservation_status BEFORE
update on rental_reservations for EACH row
execute FUNCTION check_reservation_status ();

create table public.customer_orders (
  id integer generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  order_number bigint null,
  customer_infos jsonb null,
  total_price real null,
  constraint customer_orders_pkey primary key (id)
) TABLESPACE pg_default;
